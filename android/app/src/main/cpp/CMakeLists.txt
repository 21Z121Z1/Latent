cmake_minimum_required(VERSION 3.22.1)
project(latent_android LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LATENT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LATENT_ENABLE_VULKAN_RUNTIME ON CACHE BOOL "" FORCE)
set(LATENT_STRICT_WARNINGS ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." "${CMAKE_CURRENT_BINARY_DIR}/latent")
add_library(latent_android SHARED NativePipeline.cpp HighResolutionNative.cpp)
target_link_libraries(latent_android PRIVATE latent::core jnigraphics log)
target_compile_features(latent_android PRIVATE cxx_std_20)
target_compile_options(latent_android PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror -ffp-contract=off)
